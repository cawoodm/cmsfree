{{- /* cmsfree: renders the Markdown block NAME.md (content/[/]blocks/NAME.md) between edit markers. A block missing in the page's language is a build error (no fallback). Usage: {{ partial "cms-block.html" "HERO.md" }} */ -}} {{- $name := . -}} {{- $path := printf "/blocks/%s" (strings.TrimSuffix ".md" $name) -}} {{- $p := site.GetPage $path -}} {{- if not $p }}{{ errorf "%s: no block %s (expected %s)" site.Language.Lang $name (cond hugo.IsMultilingual (printf "content/%s/blocks/%s" site.Language.Lang $name) (printf "content/blocks/%s" $name)) }}{{ end -}} {{- printf "" $name | safeHTML }}{{ $p.Content | strings.TrimSpace | safeHTML }}{{ "" | safeHTML -}}
