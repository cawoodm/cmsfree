{{- /* cmsfree: renders the block NAME.md split at "### " headings, one

    • per part, between edit markers. Fallback as in cms-block.html. Usage:
      {{ partial "cms-items.html" "SERVICES.md" }}
    */ -}} {{- $name := . -}} {{- $path := printf "/blocks/%s" (strings.TrimSuffix ".md" $name) -}} {{- $p := site.GetPage $path -}} {{- if not $p }}{{ $p = site.Sites.Default.GetPage $path }}{{ end -}} {{- if not $p }}{{ errorf "%s: no block %s" site.Language.Lang $name }}{{ end -}} {{- $parts := split (replaceRE `(?m)^### ` "\x00### " $p.RawContent) "\x00" -}} {{- $items := slice -}} {{- range $parts }}{{ with strings.TrimSpace . }}{{ $items = $items | append (printf "
