{{- /* cmsfree: renders the Markdown block NAME.md (content/blocks/NAME.md, or NAME..md in a non-default language) between edit markers. A block missing in the page's language is a build error. Usage: {{ partial "cms-block.html" "HERO.md" }} */ -}} {{- $name := . -}} {{- $path := printf "/blocks/%s" (strings.TrimSuffix ".md" $name) -}} {{- $p := site.GetPage $path -}} {{- if not $p }}{{ errorf "%s: no block %s (expected content/blocks/%s)" site.Language.Lang $name (cond (eq site.Language.Lang site.Sites.Default.Language.Lang) $name (printf "%s.%s.md" (strings.TrimSuffix ".md" $name) site.Language.Lang)) }}{{ end -}} {{- printf "" $name | safeHTML }}{{ $p.Content | strings.TrimSpace | safeHTML }}{{ "" | safeHTML -}}
